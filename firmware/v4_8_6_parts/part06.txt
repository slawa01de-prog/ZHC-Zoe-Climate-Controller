      return;
    }
  }

  if (otaInProgress)
  {
    uint8_t v = triangleWave255(700);
    setRGB(0, 0, v);
    return;
  }

  if (faultLatched || currentMode == MODE_FAILSAFE)
  {
    if (blinkPhase(250)) setRGB(255, 0, 0);
    else rgbOff();
    return;
  }

  if (activeWarning != WARNING_NONE)
  {
    if (blinkPhase(500)) setRGB(255, 120, 0);
    else rgbOff();
    return;
  }

  if (serviceOutputTestActive)
  {
    uint8_t v = triangleWave255(900);
    setRGB(v, 0, v);
    return;
  }

  switch (currentMode)
  {
    case MODE_OEM:
    {
      // OEM hat keine aktive Heizstufe. Ruhiges blaues Atmen.
      uint8_t v = triangleWave255(3500);
      setRGB(0, 0, v);
      break;
    }

    case MODE_AUTO:
      applyStagePulseToBase(0, 120, 0, activeStage);
      break;

    case MODE_BOOST:
      applyStagePulseToBase(150, 70, 0, activeStage);
      break;

    case MODE_MANUAL:
      applyStagePulseToBase(0, 90, 130, activeStage);
      break;

    case MODE_FAILSAFE:
    default:
      rgbRed();
      break;
  }
}

void printFaultHistory()
{
  Serial.println();
  Serial.println("============== FEHLERHISTORIE =================");

  if (faultHistoryCount == 0)
  {
    Serial.println("Keine Fehler gespeichert.");
    Serial.println("================================================");
    return;
  }

  for (uint8_t n = 0; n < faultHistoryCount; n++)
  {
    int index =
      faultHistoryIndex - 1 - n;

    while (index < 0)
      index += FAULT_HISTORY_SIZE;

    FaultRecord &f =
      faultHistory[index];

    Serial.print("#");
    Serial.print(n + 1);

    Serial.print(" ");
    Serial.print(faultName(f.code));

    Serial.print(" | t=");
    Serial.print(f.timestamp / 1000UL);
    Serial.print("s");

    Serial.print(" | LOW=");
    Serial.print(f.lowTemp);

    Serial.print(" | HIGH=");
    Serial.print(f.highTemp);

    Serial.print(" | CTRL=");
    Serial.print(f.controlTemp);

    Serial.print(" | OUT=");
    Serial.print(f.outsideTemp);

    Serial.print(" | SAFE=");
    Serial.print(f.safetyTemp);

    Serial.print(" | STAGE=");
    Serial.println(f.stage);
  }

  Serial.println("================================================");
}

bool currentSafetyConditionsOkay()
{
  scanSensors();

  if (!configValid())
    return false;

  if (!anyCabinRoleAssigned())
    return false;

  // Mindestens einer muss aktuell vorhanden sein.
  bool lowPresent =
    sensorLow.assigned && sensorLow.present;

  bool highPresent =
    sensorHigh.assigned && sensorHigh.present;

  if (!lowPresent && !highPresent)
    return false;

  // Wenn SAFETY zugewiesen ist, muss er vorhanden sein.
  if (sensorSafety.assigned && !sensorSafety.present)
    return false;

  return true;
}

void resetFault()
{
  if (!faultLatched)
  {
    Serial.println("Kein verriegelter Fehler vorhanden.");
    return;
  }

  if (!currentSafetyConditionsOkay())
  {
    Serial.println("FAULT RESET NICHT MOEGLICH.");
    Serial.println("Fehlerursache / Sensorzuordnung noch ungueltig.");
    return;
  }

  stopServiceOutputTest();
  forceOutputsOff();

  activeFault = FAULT_NONE;
  faultLatched = false;
  faultUnread = false;

  requestedStage = 0;
  activeStage = 0;

  integralTerm = 0.0f;
  targetHoldActive = false;

  currentMode = MODE_OEM;

  Serial.println("Fehler geloescht.");
  Serial.println("System bleibt im OEM-Modus.");
}
