void startServiceRgbTest(uint8_t color)
{

    rgbOff();
  }
}

void handleServiceAction()
{
  if (!requireAuthentication())
    return;

  String action = webServer.arg("action");
  uint32_t durationSec = (uint32_t)webServer.arg("duration").toInt();

  bool okay = true;
  String message = "OK";

  if (action == "h1")
  {
    okay = startServiceOutputTest(0x01, durationSec);
    message = okay ? "H1-Test gestartet." : "H1-Test gesperrt oder Dauer ungueltig.";
  }
  else if (action == "h2")
  {
    okay = startServiceOutputTest(0x02, durationSec);
    message = okay ? "H2-Test gestartet." : "H2-Test gesperrt oder Dauer ungueltig.";
  }
  else if (action == "h3")
  {
    okay = startServiceOutputTest(0x04, durationSec);
    message = okay ? "H3-Test gestartet." : "H3-Test gesperrt oder Dauer ungueltig.";
  }
  else if (action == "outputs_off")
  {
    stopServiceOutputTest();
    message = "H1/H2/H3 AUS.";
  }
  else if (action == "rgb_r")
  {
    startServiceRgbTest(1);
    message = "RGB Rot.";
  }
  else if (action == "rgb_g")
  {
    startServiceRgbTest(2);
    message = "RGB Gruen.";
  }
  else if (action == "rgb_b")
  {
    startServiceRgbTest(3);
    message = "RGB Blau.";
  }
  else if (action == "rgb_off")
  {
    serviceRgbTestActive = false;
    serviceRgbTestColor = 0;
    rgbOff();
    message = "RGB AUS.";
  }
  else if (action == "scan")
  {
    scanSensors();
    message = "1-Wire neu gescannt.";
  }
  else
  {
    okay = false;
    message = "Unbekannte Service-Aktion.";
  }

  Serial.print("[SERVICE] ");
  Serial.println(message);

  String json = "{";
  json += "\"ok\":" + jsonBool(okay) + ",";
  json += "\"message\":\"" + htmlEscape(message) + "\"";
  json += "}";

  webServer.sendHeader("Cache-Control", "no-store");
  webServer.send(okay ? 200 : 400, "application/json; charset=utf-8", json);
}

void setRoleFromBackup(SensorRole &role, const String &value)
{
  if (value == "NONE")
  {
    role.assigned = false;
    role.present = false;
    role.valid = false;
    return;
  }

  DeviceAddress a;
  if (stringToAddress(value, a))
  {
    memcpy(role.address, a, 8);
    role.assigned = true;
    role.present = false;
    role.valid = false;
  }
}

bool restoreConfigBackup(String text)
{
  text.replace("\r", "");

  if (!text.startsWith("ZHC_CONFIG_V1\n"))
    return false;

  Settings oldCfg = cfg;
  ThermalModel oldModel = model;
  SensorRole oldLow = sensorLow, oldHigh = sensorHigh, oldOutside = sensorOutside, oldSafety = sensorSafety;

  int pos = 0;
  while (pos < text.length())
  {
    int eol = text.indexOf('\n', pos);
    if (eol < 0) eol = text.length();

    String line = text.substring(pos, eol);
    pos = eol + 1;

    int eq = line.indexOf('=');
    if (eq <= 0) continue;

    String k = line.substring(0, eq);
    String v = line.substring(eq + 1);
    v.trim();

    if (k == "target") cfg.targetTemp = v.toFloat();
    else if (k == "deadband") cfg.deadBand = v.toFloat();
    else if (k == "restart") cfg.restartDelta = v.toFloat();
    else if (k == "wLow") cfg.weightLow = v.toFloat();
    else if (k == "wHigh") cfg.weightHigh = v.toFloat();
    else if (k == "kp") cfg.kp = v.toFloat();
    else if (k == "ki") cfg.ki = v.toFloat();
    else if (k == "trend") cfg.trendGain = v.toFloat();
    else if (k == "st1") cfg.stage1Threshold = v.toFloat();
    else if (k == "st2") cfg.stage2Threshold = v.toFloat();
    else if (k == "st3") cfg.stage3Threshold = v.toFloat();
    else if (k == "mintime") cfg.minStageTimeSec = (uint32_t)v.toInt();
    else if (k == "delay") cfg.stageDelaySec = (uint32_t)v.toInt();
    else if (k == "boost") cfg.boostTimeSec = (uint32_t)v.toInt();
    else if (k == "learning") cfg.learningEnabled = v.toInt() != 0;
    else if (k == "learnrate") cfg.learningRate = v.toFloat();
    else if (k == "learntime") cfg.learningMinTimeSec = (uint32_t)v.toInt();
    else if (k == "gain1") model.gainStage1 = v.toFloat();
    else if (k == "gain2") model.gainStage2 = v.toFloat();
    else if (k == "gain3") model.gainStage3 = v.toFloat();
    else if (k == "coast") model.coastTemp = v.toFloat();
    else if (k == "emergency") cfg.emergencyMaxTemp = v.toFloat();
    else if (k == "sensTO") cfg.sensorTimeoutSec = (uint32_t)v.toInt();
    else if (k == "tempjump") cfg.maxTempJump = v.toFloat();
    else if (k == "safeDiff") cfg.safetyMaxDifference = v.toFloat();
    else if (k == "safeDelay") cfg.safetyDifferenceDelaySec = (uint32_t)v.toInt();
    else if (k == "lowAdr") setRoleFromBackup(sensorLow, v);
    else if (k == "highAdr") setRoleFromBackup(sensorHigh, v);
    else if (k == "outAdr") setRoleFromBackup(sensorOutside, v);
    else if (k == "safeAdr") setRoleFromBackup(sensorSafety, v);
    else if (k == "lowOff") sensorLow.offset = v.toFloat();
    else if (k == "highOff") sensorHigh.offset = v.toFloat();
    else if (k == "outOff") sensorOutside.offset = v.toFloat();
    else if (k == "safeOff") sensorSafety.offset = v.toFloat();
  }

  if (!configValid())
  {
    cfg = oldCfg;
    model = oldModel;
    sensorLow = oldLow;
    sensorHigh = oldHigh;
    sensorOutside = oldOutside;
    sensorSafety = oldSafety;
    return false;
  }

  saveSettings();
  scanSensors();

  requestedStage = 0;
  activeStage = 0;
  integralTerm = 0.0f;
  controllerOutput = 0.0f;
  forceOutputsOff();
  currentMode = MODE_OEM;
  return true;
}

            "<form method='POST' action='/update' enctype='multipart/form-data'>"
            "<input type='file' name='firmware' accept='.bin'>"
            "<button type='submit'>Update starten</button>"
            "</form></div>");

  html += pageFooter();

  webServer.send(200, "text/html; charset=utf-8", html);
}

void handleUpdateUpload()
{
  if (!isAuthenticated() || forcePasswordChange)
  {
    otaUploadAuthorized = false;
    return;
  }

  otaUploadAuthorized = true;

  HTTPUpload &upload = webServer.upload();

  if (upload.status == UPLOAD_FILE_START)
  {
    otaInProgress = true;

    // Vor OTA: eigene Regelung in sicheren Zustand.
    currentMode = MODE_OEM;
    requestedStage = 0;
    activeStage = 0;
    controllerOutput = 0.0f;
    integralTerm = 0.0f;
    stopServiceOutputTest();
    forceOutputsOff();

    Serial.println();
    Serial.println("[OTA] Update gestartet.");
    Serial.println("[OTA] Eigene Heizanforderung AUS / OEM.");

    if (!Update.begin(UPDATE_SIZE_UNKNOWN))
    {
      Update.printError(Serial);
    }
  }
  else if (upload.status == UPLOAD_FILE_WRITE)
  {
    if (Update.write(upload.buf, upload.currentSize) != upload.currentSize)
    {
      Update.printError(Serial);
    }
  }
  else if (upload.status == UPLOAD_FILE_END)
  {
    if (Update.end(true))
    {
      Serial.print("[OTA] Erfolgreich. Bytes: ");
      Serial.println(upload.totalSize);
    }
    else
    {
      Update.printError(Serial);
    }

    otaInProgress = false;
  }
  else if (upload.status == UPLOAD_FILE_ABORTED)
  {
    Update.abort();
    otaInProgress = false;
    Serial.println("[OTA] Abgebrochen.");
  }
}

void handleUpdateFinish()
{
  if (!otaUploadAuthorized)
  {
    webServer.send(401, "text/plain; charset=utf-8",
                   "Nicht autorisiert.");
    return;
  }

  if (Update.hasError())
  {
    webServer.send(500, "text/plain; charset=utf-8",
                   "Firmware-Update fehlgeschlagen.");
    return;
  }

  webServer.send(
    200,
    "text/html; charset=utf-8",
    "<!doctype html><html><body><h2>Update erfolgreich</h2>"
    "<p>Controller startet neu.</p></body></html>"
  );

  delay(500);
  forceOutputsOff();
  ESP.restart();
}

void handleLogout()
{
  invalidateWebSession();

  webServer.sendHeader(
    "Set-Cookie",
    "ZHCSESSION=; HttpOnly; SameSite=Strict; Max-Age=0; Path=/"
  );

  webServer.sendHeader("Location", "/login");
  webServer.send(302, "text/plain", "");
}

void configureWebServer()
{
  const char* headerKeys[] = { "Cookie" };
  webServer.collectHeaders(headerKeys, 1);

  webServer.on("/login", HTTP_GET, handleLoginPage);
  webServer.on("/login", HTTP_POST, handleLoginPage);

  webServer.on("/password", HTTP_GET, handlePasswordPage);
  webServer.on("/password", HTTP_POST, handlePasswordPage);

  webServer.on("/", HTTP_GET, handleRootPage);

  webServer.on("/control", HTTP_GET, handleControlPage);
  webServer.on("/control", HTTP_POST, handleControlPage);

  webServer.on("/sensors", HTTP_GET, handleSensorsPage);
  webServer.on("/sensors", HTTP_POST, handleSensorsPage);

  webServer.on("/regulator", HTTP_GET, handleRegulatorPage);
  webServer.on("/regulator", HTTP_POST, handleRegulatorPage);

  webServer.on("/learning", HTTP_GET, handleLearningPage);
  webServer.on("/learning", HTTP_POST, handleLearningPage);

  webServer.on("/diagnostics", HTTP_GET, handleDiagnosticsPage);
  webServer.on("/diagnostics", HTTP_POST, handleDiagnosticsPage);

  webServer.on("/service", HTTP_GET, handleServicePage);
  webServer.on("/service/status", HTTP_GET, handleServiceStatusJson);
  webServer.on("/service/action", HTTP_POST, handleServiceAction);

  webServer.on("/system", HTTP_GET, handleSystemPage);
  webServer.on("/backup", HTTP_GET, handleBackupPage);
  webServer.on("/restore", HTTP_GET, handleRestorePage);
  webServer.on("/restore", HTTP_POST, handleRestorePage);

  webServer.on("/update", HTTP_GET, handleUpdatePage);
  webServer.on(
    "/update",
    HTTP_POST,
    handleUpdateFinish,
    handleUpdateUpload
  );

  webServer.on("/logout", HTTP_GET, handleLogout);

  webServer.onNotFound([]()
  {
    if (!isAuthenticated())
    {
      webServer.sendHeader("Location", "/login");
      webServer.send(302, "text/plain", "");
      return;
    }

    webServer.send(404, "text/plain; charset=utf-8", "Nicht gefunden.");
  });
}

void saveWiFiSettings()
{
  preferences.begin("zhcwifi", false);
  preferences.putBool("startup", wifiStartupEnabled);
  preferences.putString("apPwd", apPassword);
  preferences.end();
}

void loadWiFiSettings()
{
  preferences.begin("zhcwifi", true);

  wifiStartupEnabled =
    preferences.getBool("startup", false);

  apPassword =
    preferences.getString("apPwd", "");

  preferences.end();

  if (apPassword.length() < 8)
  {
    apPassword = "ZHC-" + randomToken(8);
    saveWiFiSettings();
  }
}

  Serial.print("SSID           : ");
  Serial.println(ZHC_AP_SSID);

  Serial.print("AP-Passwort    : ");
  Serial.println(apPassword);

  Serial.print("Webserver      : ");
  Serial.println(webServerRunning ? "AKTIV" : "AUS");

  if (wifiEnabled)
  {
    Serial.print("IP             : ");
    Serial.println(WiFi.softAPIP());
    Serial.println("Web            : http://192.168.4.1");
  }

  Serial.println("================================================");
}
