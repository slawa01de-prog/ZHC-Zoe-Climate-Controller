    return 0;

String randomToken(size_t length)
{
  static const char alphabet[] =
    "ABCDEFGHJKLMNPQRSTUVWXYZ"
    "abcdefghijkmnopqrstuvwxyz"
    "23456789";

  String result;
  result.reserve(length);

  for (size_t i = 0; i < length; i++)
  {
    uint32_t r = esp_random();
    result += alphabet[r % (sizeof(alphabet) - 1)];
  }

  return result;
}

String randomHex(size_t bytes)
{
  static const char hex[] = "0123456789ABCDEF";

  String result;
  result.reserve(bytes * 2);

  for (size_t i = 0; i < bytes; i++)
  {
    uint8_t b = (uint8_t)(esp_random() & 0xFF);
    result += hex[(b >> 4) & 0x0F];
    result += hex[b & 0x0F];
  }

  return result;
}

String sha256Hex(const String &input)
{
  unsigned char digest[32];

  mbedtls_sha256_context ctx;
  mbedtls_sha256_init(&ctx);

  mbedtls_sha256_starts(&ctx, 0);
  mbedtls_sha256_update(
    &ctx,
    reinterpret_cast<const unsigned char*>(input.c_str()),
    input.length()
  );
  mbedtls_sha256_finish(&ctx, digest);
  mbedtls_sha256_free(&ctx);

  static const char hex[] = "0123456789abcdef";
  String out;
  out.reserve(64);

  for (uint8_t i = 0; i < 32; i++)
  {
    out += hex[(digest[i] >> 4) & 0x0F];
    out += hex[digest[i] & 0x0F];
  }

  return out;
}

String makePasswordHash(const String &password)
{
  return sha256Hex(authSalt + ":" + password);
}

bool constantTimeStringEqual(const String &a, const String &b)
{
  if (a.length() != b.length())
    return false;

  uint8_t diff = 0;

  for (size_t i = 0; i < a.length(); i++)
    diff |= (uint8_t)(a[i] ^ b[i]);

  return diff == 0;
}

String htmlEscape(const String &value)
{
  String out;
  out.reserve(value.length() + 16);

  for (size_t i = 0; i < value.length(); i++)
  {
    switch (value[i])
    {
      case '&': out += "&amp;"; break;
      case '<': out += "&lt;"; break;
      case '>': out += "&gt;"; break;
      case '"': out += "&quot;"; break;
      case '\'': out += "&#39;"; break;
      default: out += value[i]; break;
    }
  }

  return out;
}

String getCookieValue(const String &name)
{
  if (!webServer.hasHeader("Cookie"))
    return "";

  String cookie = webServer.header("Cookie");
  String key = name + "=";

  int start = cookie.indexOf(key);
  if (start < 0)
    return "";

  start += key.length();

  int end = cookie.indexOf(';', start);
  if (end < 0)
    end = cookie.length();

  return cookie.substring(start, end);
}

bool isAuthenticated()
{
  if (sessionToken.length() == 0)
    return false;

  if (millis() - sessionLastActivityMillis > WEB_SESSION_TIMEOUT_MS)
  {
    invalidateWebSession();
    return false;
  }

  String cookieToken = getCookieValue("ZHCSESSION");

  if (!constantTimeStringEqual(cookieToken, sessionToken))
    return false;

  sessionLastActivityMillis = millis();
  return true;
}

bool requireAuthentication(bool allowForcedPasswordPage = false)
{
  if (!isAuthenticated())
  {
    webServer.sendHeader("Location", "/login");
    webServer.send(302, "text/plain", "");
    return false;
  }

  if (forcePasswordChange && !allowForcedPasswordPage)
  {
    webServer.sendHeader("Location", "/password");
    webServer.send(302, "text/plain", "");
    return false;
  }

  return true;
}

{
  if (webServer.method() == HTTP_GET)
  {

        "<meta name='viewport' content='width=device-width,initial-scale=1'>"

        ".card{background:#1d1d1d;padding:24px;border-radius:12px}"

        "button{padding:10px 16px}</style></head><body>"
        "<div class='card'><h2>ZOE Climate Controller</h2>"
        "<form method='post' action='/login'>"
        "<label>Benutzer</label><input name='user' value='admin' autocomplete='username'>"
        "<label>Passwort</label><input type='password' name='pass' autocomplete='current-password'>"
        "<button type='submit'>Anmelden</button>"
        "</form></div></body></html>");

    webServer.send(200, "text/html; charset=utf-8", html);
    return;
  }

  if (isLoginLocked())
  {
    webServer.send(429, "text/plain; charset=utf-8",
                   "Zu viele Fehlversuche. Bitte spaeter erneut versuchen.");
    return;
  }

  String user = webServer.arg("user");
  String pass = webServer.arg("pass");

  bool userOkay = (user == "admin");
  bool passOkay =
    constantTimeStringEqual(makePasswordHash(pass), passwordHash);

  if (!userOkay || !passOkay)
  {
    failedLoginCount++;

    if (failedLoginCount >= MAX_LOGIN_FAILURES)
    {
      loginLockoutUntilMillis = millis() + LOGIN_LOCKOUT_MS;
      failedLoginCount = 0;
    }

    webServer.send(401, "text/plain; charset=utf-8",
                   "Login fehlgeschlagen.");
    return;
  }

  failedLoginCount = 0;
  loginLockoutUntilMillis = 0;

  sessionToken = randomHex(24);
  sessionLastActivityMillis = millis();

  webServer.sendHeader(
    "Set-Cookie",
    "ZHCSESSION=" + sessionToken +
      "; HttpOnly; SameSite=Strict; Max-Age=900; Path=/"
  );

  webServer.sendHeader(
    "Location",
    forcePasswordChange ? "/password" : "/"
  );

  webServer.send(302, "text/plain", "");
}

    cfg.ki = webServer.arg("ki").toFloat();
    cfg.trendGain = webServer.arg("trend").toFloat();
    cfg.stage1Threshold = webServer.arg("st1").toFloat();
    cfg.stage2Threshold = webServer.arg("st2").toFloat();
    cfg.stage3Threshold = webServer.arg("st3").toFloat();
    cfg.minStageTimeSec = (uint32_t)webServer.arg("mintime").toInt();
    cfg.stageDelaySec = (uint32_t)webServer.arg("delay").toInt();
    cfg.boostTimeSec = (uint32_t)webServer.arg("boost").toInt();
    cfg.safetyMaxDifference = webServer.arg("safediff").toFloat();
    cfg.safetyDifferenceDelaySec = (uint32_t)webServer.arg("safedelay").toInt();
    cfg.emergencyMaxTemp = webServer.arg("emergency").toFloat();
    cfg.sensorTimeoutSec = (uint32_t)webServer.arg("sensorto").toInt();
    cfg.maxTempJump = webServer.arg("tempjump").toFloat();

    if (!configValid())
    {
      cfg = oldCfg;
      webServer.send(400, "text/plain; charset=utf-8", "Ungueltige Reglerkonfiguration.");
      return;
    }

    saveSettings();
    webServer.sendHeader("Location", "/regulator");
    webServer.send(303, "text/plain", "");
    return;
  }

  String html = pageHeader("Regler");
  html += F("<div class='card'><h2>PI / Heizstufen / Sicherheit</h2><form method='post' action='/regulator'>");

  html += "<label>Kp</label><input name='kp' type='number' step='0.1' value='" + String(cfg.kp, 2) + "'>";
  html += "<label>Ki</label><input name='ki' type='number' step='0.01' value='" + String(cfg.ki, 3) + "'>";
  html += "<label>Trend Gain</label><input name='trend' type='number' step='0.1' value='" + String(cfg.trendGain, 2) + "'>";

  html += "<label>Stufe 1 Schwelle [%]</label><input name='st1' type='number' step='1' value='" + String(cfg.stage1Threshold, 1) + "'>";
  html += "<label>Stufe 2 Schwelle [%]</label><input name='st2' type='number' step='1' value='" + String(cfg.stage2Threshold, 1) + "'>";
  html += "<label>Stufe 3 Schwelle [%]</label><input name='st3' type='number' step='1' value='" + String(cfg.stage3Threshold, 1) + "'>";

  html += "<label>Mindestzeit Stufe [s]</label><input name='mintime' type='number' value='" + String(cfg.minStageTimeSec) + "'>";
  html += "<label>Stufenverzoegerung [s]</label><input name='delay' type='number' value='" + String(cfg.stageDelaySec) + "'>";
  html += "<label>Boost-Zeit [s]</label><input name='boost' type='number' value='" + String(cfg.boostTimeSec) + "'>";

  html += "<label>SAFETY max. Abweichung [K]</label><input name='safediff' type='number' step='0.1' value='" + String(cfg.safetyMaxDifference, 2) + "'>";
  html += "<label>SAFETY Verzoegerung [s]</label><input name='safedelay' type='number' value='" + String(cfg.safetyDifferenceDelaySec) + "'>";
  html += "<label>Not-Aus Temperatur [&deg;C]</label><input name='emergency' type='number' step='0.5' value='" + String(cfg.emergencyMaxTemp, 1) + "'>";
  html += "<label>Sensor Timeout [s]</label><input name='sensorto' type='number' value='" + String(cfg.sensorTimeoutSec) + "'>";
  html += "<label>Max. Temperatursprung [K]</label><input name='tempjump' type='number' step='0.1' value='" + String(cfg.maxTempJump, 1) + "'>";

  html += F("<button type='submit'>Regler speichern</button></form></div>");
  html += pageFooter();
  webServer.send(200, "text/html; charset=utf-8", html);
}
